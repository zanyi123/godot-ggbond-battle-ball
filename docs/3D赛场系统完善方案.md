## 四、核心代码模块

### 4.1 `ball_proxy_3d.gd`（新文件）

球 3D 代理脚本，挂在 BallProxy 节点上。

**功能**：
1. 加载 battleball.glb
2. 同步 2D ball.gd 的 position
3. 同步 is_active 状态（飞行/持球）
4. 飞行时绕飞行方向轴自旋
5. 接球时停靠到 player 的 hand_proxy

**关键代码框架**：
```gdscript
extends Node3D

const BALL_BODY_PATH := "res://建模素材库/3D模型素材/battleball.glb"
const BALL_SCALE: float = 70.0  # 与球员匹配
const BALL_FLIGHT_HEIGHT: float = 30.0  # 飞行高度
const BALL_CARRIED_HEIGHT: float = 70.0  # 持球高度
const BALL_SPIN_SPEED: float = 8.0  # 自旋速度 (rad/s)

@export_node_path var ball_2d: NodePath  # 2D ball.gd 节点

var ball_2d_node: Area2D = null
var ball_mesh: MeshInstance3D = null
var is_flying: bool = false
var flight_direction: Vector2 = Vector2.RIGHT

func _ready() -> void:
    ball_2d_node = get_node(ball_2d)
    _load_ball_model()

func _load_ball_model() -> void:
    var ball_scene: PackedScene = load(BALL_BODY_PATH)
    if ball_scene == null:
        push_error("[BallProxy3D] 加载失败: %s" % BALL_BODY_PATH)
        return
    var ball_inst: Node3D = ball_scene.instantiate()
    if ball_inst and ball_inst is Node3D:
        add_child(ball_inst)
        ball_mesh = ball_inst.get_node_or_null("BallMesh") as MeshInstance3D
        # 缩放匹配球员
        ball_inst.scale = Vector3(BALL_SCALE, BALL_SCALE, BALL_SCALE)
        # 修复 PBR
        _fix_pbr(ball_inst)
        # 隐藏 Mixamo 残留
        _hide_mixamo_helpers(ball_inst)

func _process(delta: float) -> void:
    if ball_2d_node == null:
        return
    # 同步 2D → 3D 位置
    var pos2d: Vector2 = ball_2d_node.global_position
    var pos3d: Vector3 = Vector3(pos2d.x, _get_ball_y(), pos2d.y)
    global_position = pos3d
    # 飞行时自旋
    is_flying = ball_2d_node.is_active
    if is_flying:
        # 绕飞行方向轴旋转
        var axis: Vector3 = Vector3(flight_direction.x, 0, flight_direction.y).normalized()
        rotate(axis, BALL_SPIN_SPEED * delta)
        flight_direction = ball_2d_node.ball_direction
    else:
        # 持球时跟随手
        _attach_to_hand()

func _get_ball_y() -> float:
    if ball_2d_node.is_active:
        return BALL_FLIGHT_HEIGHT
    elif ball_2d_node.owner_player != null:
        return BALL_CARRIED_HEIGHT
    return 0.0

func _attach_to_hand() -> void:
    # 找持球球员的 hand_proxy
    var owner = ball_2d_node.owner_player
    if owner == null or not is_instance_valid(owner):
        return
    # 队伍 A → proxy_a, 队伍 B → proxy_b
    var proxy_root = _find_player_proxy(owner)
    if proxy_root == null:
        return
    var hand_proxy = proxy_root.get_node_or_null("HandProxy")
    if hand_proxy != null:
        # 球跟随 hand_proxy
        global_position = hand_proxy.global_position
```

## 五、动作与球适配

### 5.1 投球（throw 动作）
- F1 按下 → 播 throw 动画
- 球在 throw 动画期间**一直粘在手上**（hand_proxy）
- throw 动画到达"出手帧"（约 0.5-0.7s）→ 球从 hand_proxy 脱离 → ball.gd.launch()
- 球脱离后开始自旋飞行
- throw 动画播完 → 切回 idle

**实现要点**：
- ball.gd 的 `launch()` 之前是手动调用
- 新增 `BallProxy3D` 检测 throw 动画的"出手帧"
- 出手帧 = Animation 的 "fire" track（新增的 track）
- 或者：用 throw 动画的 0.6 进度作为出手点

### 5.2 接球（catch 动作）
- ball.gd 飞行 → 检测碰撞 → `ball_caught` 信号
- BallProxy3D 监听信号 → 球飞向 player 的 hand_proxy（追踪）
- player 播 catch 动画
- 球到达 hand_proxy 位置 → 完全粘住
- catch 动画播完 → 球变为"持球"状态

### 5.3 受击（受击动作）
- ball.gd 命中球员 → `ball_hit_player` 信号
- 球员播 hit 动画（如果有）或 catch 动画
- ball.gd 状态：`is_active = false` → 球停飞

**第二轮修复（2026-07-21 后续）**：

根因分析：自检发现 FBX mesh AABB 仅 (0.01, 0.01, 0.007)，而 GLB AABB 为 (1.08, 0.71, 1.08)。之前 FBX scale=10 导致全局高度仅 7 单位（正交 size=870 下不到 1% 屏幕），而 GLB 被 `visible=false` 隐藏，所以用户看不到正常大小的模型。

修复：
1. FBX `idle_inst.scale` 从 10 改为 **65**（70×65=4550，全局高度≈50 单位，与 GLB 匹配）
2. `_give_ball_to()` 接球后重置 `proxy.rotation.y = 0.0`（HandProxy 回正到头顶）
3. `_hk_idle()` 切 idle 时也重置 `rotation.y = 0.0`
4. 重写 `player_3d_test.tscn`，删除代码动态创建的重复节点（BigCameraViewport/FieldBG/PlayerA/PlayerB/DebugUI 等），只保留 Camera2D
